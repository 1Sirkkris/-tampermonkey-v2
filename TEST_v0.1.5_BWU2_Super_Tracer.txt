  function sensitiveKey(key) {

  function redactObject(value, depth = 0) {
    if (depth > 6) return '[DEPTH_LIMIT]';
    if (value == null) return value;
    if (Array.isArray(value)) return value.slice(0, 100).map(v => redactObject(v, depth + 1));
    if (typeof value === 'object') {

        out[k] = sensitiveKey(k) ? '[REDACTED]' : redactObject(v, depth + 1);
      }
      return out;
    }
    if (typeof value === 'string') {
      if (/^Bearer\s+/i.test(value) || /^AWS4-HMAC-SHA256/i.test(value)) return '[REDACTED]';
      return clip(value);
    }
    return value;
  }

  function sanitizeUrl(raw) {

        origin: u.origin,
        host: u.host,
        path: u.pathname,
        query: u.search,
        full: `${u.origin}${u.pathname}${u.search}`
      };
    } catch {
      return { origin: '', host: '', path: clip(raw, 500), query: '', full: clip(raw, 500) };
    }
  }

  function parseRequestBody(body) {
    if (body == null) return null;

      if (typeof body === 'string') {

        try { return redactObject(JSON.parse(body)); }

              const out = {};
              for (const [k, v] of p.entries()) out[k] = sensitiveKey(k) ? '[REDACTED]' : clip(v, 500);
              return out;
            }
          }
          return clip(body);
        }
      }

        const out = {};

        }
        return out;
      }
      return clip(body);
    } catch {
      return '[unreadable body]';
    }
  }

  function stableString(value) {

    if (Array.isArray(value)) return `[${value.map(stableString).join(',')}]`;
    return `{${Object.keys(value).sort().map(k => `${k}:${stableString(value[k])}`).join(',')}}`;
  }

  function fnv1a(str) {
    let h = 2166136261;
    for (let i = 0; i < str.length; i++) {
      h ^= str.charCodeAt(i);
      h = Math.imul(h, 16777619);
    }
    return (h >>> 0).toString(36);
  }

  function requestSignature(method, url, body) {

  function duplicateInfo(signature, classification = 'workflow') {
    const t = epoch();
    const previous = recentRequestSignatures.get(signature) || 0;
    recentRequestSignatures.set(signature, t);
    if (recentRequestSignatures.size > 300) {
      for (const [k, v] of recentRequestSignatures) {
        if (t - v > 10000) recentRequestSignatures.delete(k);
      }
    }
    const eligible = classification === 'workflow';
    return {
      duplicate: eligible && !!previous && t - previous <= DUP_WINDOW_MS,
      previousGapMs: previous ? t - previous : null
    };
  }

  function classifyInput(raw) {
    const value = clean(raw);
    const upper = value.toUpperCase();
    let kind = 'other';
    if (/^(?:TSX|CSX)[A-Z0-9]+$/i.test(value)) kind = 'container';
    else if (/^X[A-Z0-9]{9}$/i.test(value)) kind = 'fnsku';
    else if (/^B0[A-Z0-9]{8}$/i.test(value)) kind = 'asin';
    else if (/^\d{8,14}$/.test(value)) kind = 'numeric_barcode';
    else if (/^P-\d+-[A-Z0-9-]+$/i.test(value)) kind = 'location';
    else if (/^DZ-[A-Z0-9_-]+$/i.test(value)) kind = 'dropzone';
    else if (/^\d{1,4}$/.test(value)) kind = 'quantity';
    else if (/^\d{1,4}[\/-]\d{1,2}[\/-]\d{1,4}$/.test(value)) kind = 'date';

    return {
      kind,
      length: value.length,
      value: settings.includeRecognizedValues && kind !== 'other' ? upper : undefined
    };
  }

  function targetHint(el) {
    if (!el) return '';
    const bits = [
      el.getAttribute?.('aria-label'), el.getAttribute?.('placeholder'), el.getAttribute?.('name'),
      el.getAttribute?.('id'), el.getAttribute?.('data-testid')
    ].map(clean).filter(Boolean);
    return clip(bits.join(' | '), 240);
  }

  function installActionTrace() {
    document.addEventListener('keydown', event => {
      if (event.isTrusted) markForegroundActivity('keyboard');
      if (event.key !== 'Enter') return;
      const el = event.target;
      const tag = String(el?.tagName || '').toLowerCase();
      const isField = tag === 'input' || tag === 'textarea' || el?.isContentEditable;
      const inputSummary = isField ? classifyInput(el?.value ?? el?.textContent ?? '') : null;
      add('ENTER', {
        trusted: !!event.isTrusted,
        field: isField ? targetHint(el) : '',
        input: inputSummary
      });
      if (event.isTrusted) publishTrustedAction('enter', inputSummary);
    }, true);

    document.addEventListener('click', event => {
      if (event.isTrusted) markForegroundActivity('click');
      const el = event.target?.closest?.('button,[role="button"],input[type="button"],input[type="submit"],a');
      if (!el) return;
      const label = clean(el.innerText || el.value || el.getAttribute?.('aria-label') || el.title || '');
      if (!label) return;
      add('CLICK', {
        trusted: !!event.isTrusted,
        tag: String(el.tagName || '').toLowerCase(),
        label: clip(label, 160),
        href: el.tagName === 'A' ? clip(el.getAttribute('href') || '', 300) : ''
      });
      if (event.isTrusted) publishTrustedAction('click', { label: clip(label, 160) });
    }, true);
  }

  function limitedTextSample(root, maxChars = 12000) {
    if (!root) return '';
    try {
      const walker = document.createTreeWalker(root, NodeFilter.SHOW_TEXT);
      let out = '';
      let node;
      while ((node = walker.nextNode()) && out.length < maxChars) {
        const parent = node.parentElement;
        if (!parent || /^(SCRIPT|STYLE|NOSCRIPT|SVG)$/i.test(parent.tagName)) continue;
        const text = clean(node.nodeValue);
        if (!text) continue;
        out += `${text} `;
      }
      return out.slice(0, maxChars);
    } catch {
      return '';
    }
  }

  function knownScreenLabel() {
    const path = location.pathname;
    const title = clean(document.title);
    const headingEl = document.querySelector('h1,h2,[role="heading"]');
    const heading = clean(headingEl?.textContent || '');

    let workflow = 'Other';
    if (/aft-poirot/i.test(location.host)) workflow = 'Sideline';
    else if (/fcresearch|qifcr/i.test(location.host)) workflow = 'FCResearch';
    else if (/aft-moveapp/i.test(location.host)) workflow = 'MoveContainer';
    else if (/\/app\/edititems/i.test(path)) workflow = 'EditItems';
    else if (/\/app\/moveitems/i.test(path)) workflow = 'MoveItems';
    else if (/\/app\/fcskuflip/i.test(path)) workflow = 'FcSkuFlip';
    else if (/aft-qt/i.test(location.host)) workflow = 'AFT';

    const bodySample = limitedTextSample(document.body, 12000);
    let state = heading || title || path;
    const patterns = [
      [/scan source container/i, 'Scan source container'],
      [/scan destination container/i, 'Scan destination container'],
      [/scan container/i, 'Scan container'],
      [/verify item/i, 'Verify item'],
      [/scan item/i, 'Scan item'],
      [/enter quantity/i, 'Enter quantity'],
      [/expiration date|enter expiration/i, 'Expiration date'],
      [/remove datelot|remove date.?lot/i, 'Remove datelot'],
      [/change container/i, 'Change container'],
      [/move successful|scan another container/i, 'Move successful'],
      [/work is errored|service failed to process your request/i, 'Error'],
      [/mode\s*:\s*datelot/i, 'Mode: Datelot'],
      [/mode\s*:\s*sku/i, 'Mode: Sku'],
      [/mode\s*:\s*each/i, 'Mode: Each']
    ];
    for (const [re, label] of patterns) {
      if (re.test(bodySample)) { state = label; break; }
    }
    return `${workflow} | ${state}`;
  }

  function sampleScreen(force = false) {
    const t0 = perf();
    clearTimeout(screenTimer);
    screenTimer = 0;
    try {
      const label = knownScreenLabel();
      if (!force && label === lastScreenLabel) return;
      const previous = lastScreenLabel;
      lastScreenLabel = label;
      add('SCREEN', { label, previous, url: `${location.origin}${location.pathname}${location.search}` });
    } catch (error) {
      add('TRACER_ERROR', { area: 'screen', error: clip(error?.message || error, 800) });
    } finally {
      health.screenCalls++;
      health.screenMs += Math.max(0, perf() - t0);
    }
  }

  function scheduleScreenSample() {
    if (screenTimer) return;
    screenTimer = setTimeout(() => sampleScreen(false), SCREEN_DEBOUNCE_MS);
  }

  function installScreenTrace() {
    const start = () => {
      sampleScreen(true);
      const observer = new MutationObserver(records => {
        health.mutationCallbacks++;
        health.mutationRecords += records.length;
        for (const record of records) {
          if (record.addedNodes?.length) health.addedNodes += record.addedNodes.length;
        }
        scheduleScreenSample();
      });
      observer.observe(document.documentElement, { subtree: true, childList: true, characterData: true });
    };
    if (document.documentElement) start();
    else document.addEventListener('DOMContentLoaded', start, { once: true });
  }

  function installRouteTrace() {
    try {
      const wrap = name => {
        const original = W.history?.[name];
        if (!original || original.__bwu2SuperTraceV010) return;
        const wrapped = function(...args) {
          const before = location.href;
          const out = Reflect.apply(original, this, args);
          setTimeout(() => {
            add('ROUTE', { kind: name, before, after: location.href });
            sampleScreen(true);
          }, 0);
          return out;
        };
        wrapped.__bwu2SuperTraceV010 = true;
        W.history[name] = wrapped;
      };
      wrap('pushState');
      wrap('replaceState');
      W.addEventListener('popstate', () => { add('ROUTE', { kind: 'popstate', after: location.href }); sampleScreen(true); }, true);
      W.addEventListener('hashchange', () => { add('ROUTE', { kind: 'hashchange', after: location.href }); sampleScreen(true); }, true);
    } catch (error) {
      add('TRACER_ERROR', { area: 'route', error: clip(error?.message || error, 800) });
    }
  }

  function installErrorTrace() {
    W.addEventListener('error', event => {
      add('JS_ERROR', {
        message: clip(event?.message || 'Script error', 1000),
        source: clip(event?.filename || '', 500),
        line: event?.lineno || 0,
        column: event?.colno || 0
      });
    }, true);

    W.addEventListener('unhandledrejection', event => {
      const reason = event?.reason;
      add('PROMISE_REJECTION', { message: clip(reason?.message || reason || 'Unhandled rejection', 1000) });
    }, true);
  }

  function installPerformanceHealthTrace() {
    setInterval(() => {
      const t = epoch();
      if (!idleSince && t - lastForegroundActivityAt >= IDLE_THRESHOLD_MS) {
        idleSince = lastForegroundActivityAt + IDLE_THRESHOLD_MS;
        add('IDLE_START', {
          thresholdMs: IDLE_THRESHOLD_MS,
          idleSince
        });
      }
    }, 5000);

    let lastVisibilityChangeAt = epoch();
    document.addEventListener('visibilitychange', () => {
      lastVisibilityChangeAt = epoch();
      eventLoopExpectedAt = perf() + EVENT_LOOP_PULSE_MS;
      add('VISIBILITY', { hidden: document.hidden });
    }, true);

    eventLoopExpectedAt = perf() + EVENT_LOOP_PULSE_MS;
    setInterval(() => {
      const nowPerf = perf();
      const nowEpoch = epoch();

      if (document.hidden) {
        eventLoopExpectedAt = nowPerf + EVENT_LOOP_PULSE_MS;
        return;
      }

      const lag = Math.max(0, nowPerf - eventLoopExpectedAt);
      eventLoopExpectedAt = nowPerf + EVENT_LOOP_PULSE_MS;

      const nearVisibilityChange = nowEpoch - lastVisibilityChangeAt < 2500;
      if (lag >= TIMER_SUSPEND_CUTOFF_MS || nearVisibilityChange) {
        if (lag >= EVENT_LOOP_WARN_MS) {
          add('TIMER_SUSPEND', { ms: round(lag), nearVisibilityChange, hidden: document.hidden });
        }
        return;
      }

      health.maxEventLoopLagMs = Math.max(health.maxEventLoopLagMs, lag);
      if (lag >= EVENT_LOOP_WARN_MS) add('EVENT_LOOP_STALL', { ms: round(lag) });
    }, EVENT_LOOP_PULSE_MS);

    try {
      if (typeof PerformanceObserver === 'function') {
        const supported = PerformanceObserver.supportedEntryTypes || [];
        if (supported.includes('longtask')) {
          const po = new PerformanceObserver(list => {
            if (document.hidden) return;
            for (const entry of list.getEntries()) {
              add('LONG_TASK', {
                ms: round(entry.duration),
                startMs: round(entry.startTime),
                name: clip(entry.name || 'longtask', 100)
              });
            }
          });
          po.observe({ entryTypes: ['longtask'] });
        }
      }
    } catch {}

    setInterval(() => {
      const now = epoch();
      const windowMs = Math.max(1, now - healthWindowStartedAt);
      let domNodes = 0;
      try { domNodes = document.getElementsByTagName('*').length; } catch {}

      let heapUsedMB = null;
      let heapTotalMB = null;
      try {
        if (performance.memory) {
          heapUsedMB = Math.round(performance.memory.usedJSHeapSize / 1048576);
          heapTotalMB = Math.round(performance.memory.totalJSHeapSize / 1048576);
        }
      } catch {}

      const payload = {
        windowMs,
        visible: !document.hidden,
        domNodes,
        mutationCallbacks: health.mutationCallbacks,
        mutationRecords: health.mutationRecords,
        addedNodes: health.addedNodes,
        maxEventLoopLagMs: round(health.maxEventLoopLagMs),
        tracerAddCalls: health.addCalls,
        tracerAddMs: Math.round(health.addMs * 100) / 100,
        tracerFlushCalls: health.flushCalls,
        tracerFlushMs: Math.round(health.flushMs * 100) / 100,
        tracerScreenCalls: health.screenCalls,
        tracerScreenMs: Math.round(health.screenMs * 100) / 100,
        heapUsedMB,
        heapTotalMB
      };

      add('HEALTH_SAMPLE', payload);
      resetHealthWindow();
    }, HEALTH_SAMPLE_MS);
  }

  function allEventsForSession(sessionId) {
    flushPage();
    const keys = gmKeys().filter(k => k.startsWith(`${PAGE_PREFIX}${sessionId}:`));
    const all = [];
    for (const key of keys) {
      const page = parseJSON(gmGet(key, ''), null);
      if (!page?.events || !Array.isArray(page.events)) continue;
      all.push(...page.events);
    }
    all.sort((a, b) => (a.ts || 0) - (b.ts || 0) || String(a.id).localeCompare(String(b.id)));
    const seen = new Set();
    return all.filter(ev => {
      if (!ev?.id || seen.has(ev.id)) return false;
      seen.add(ev.id);
      return true;
    });
  }

  function percentile(values, q) {
    if (!values.length) return 0;
    const a = [...values].sort((x, y) => x - y);
    const idx = Math.max(0, Math.min(a.length - 1, Math.ceil(q * a.length) - 1));
    return round(a[idx]);
  }

  function median(values) {
    if (!values.length) return 0;
    const a = [...values].sort((x, y) => x - y);
    const mid = Math.floor(a.length / 2);
    return round(a.length % 2 ? a[mid] : (a[mid - 1] + a[mid]) / 2);
  }

  function endpointStats(events) {
    const reqById = new Map();
    const groups = new Map();
    for (const ev of events) {
      if (ev.type === 'REQUEST') reqById.set(ev.data?.reqId, ev.data);
      if (ev.type !== 'RESPONSE') continue;
      const d = ev.data || {};
      const req = reqById.get(d.reqId);
      const classification = d.class || req?.class || requestClass(d);
      const key = endpointKey(d);
      if (!groups.has(key)) {
        groups.set(key, {
          key, durations: [], count: 0, errors: 0, duplicates: 0,
          classCounts: { workflow: 0, poll: 0, background: 0 }
        });
      }
      const g = groups.get(key);
      g.count++;
      g.durations.push(Number(d.ms || 0));
      if (!d.ok || Number(d.status || 0) >= 400) g.errors++;
      if (d.duplicate || req?.duplicate) g.duplicates++;
      if (g.classCounts[classification] != null) g.classCounts[classification]++;
    }
    return [...groups.values()].map(g => {
      const mainClass = Object.entries(g.classCounts).sort((a, b) => b[1] - a[1])[0]?.[0] || 'workflow';
      return {
        endpoint: g.key,
        class: mainClass,
        count: g.count,
        medianMs: median(g.durations),
        p95Ms: percentile(g.durations, 0.95),
        maxMs: Math.max(0, ...g.durations.map(round)),
        errors: g.errors,
        duplicates: g.duplicates
      };
    }).sort((a, b) => b.count - a.count || b.p95Ms - a.p95Ms);
  }

  function actionToRequestStats(events) {
    const bestByAction = new Map();
    for (const ev of events) {
      if (ev.type !== 'REQUEST') continue;
      const d = ev.data || {};
      if ((d.class || requestClass(d)) !== 'workflow') continue;
      if (!d.actionId || !Number.isFinite(Number(d.actionGapMs))) continue;
      const gapMs = Number(d.actionGapMs);
      if (gapMs < 0 || gapMs > ACTION_CORRELATION_MS) continue;

      const current = bestByAction.get(d.actionId);
      if (!current || gapMs < current.gapMs) {
        bestByAction.set(d.actionId, {
          gapMs,
          action: d.actionKind || 'action',
          endpoint: endpointKey(d),
          at: ev.at,
          actionPageId: d.actionPageId || null,
          requestPageId: ev.pageId,
          crossedPage: !!d.actionPageId && d.actionPageId !== ev.pageId
        });
      }
    }
    return [...bestByAction.values()];
  }

  function idleStats(events, fromTs, toTs) {
    const openByPage = new Map();
    const intervals = [];
    const ordered = [...events].sort((a, b) => (a.ts || 0) - (b.ts || 0));

    for (const ev of ordered) {
      if (ev.type === 'IDLE_START') {
        if (!openByPage.has(ev.pageId)) {
          openByPage.set(ev.pageId, Number(ev.data?.idleSince || ev.ts || 0));
        }
      } else if (ev.type === 'IDLE_END') {
        const start = openByPage.get(ev.pageId);
        if (start) {
          intervals.push([Math.max(fromTs, start), Math.min(toTs, Number(ev.ts || toTs))]);
          openByPage.delete(ev.pageId);
        }
      }
    }
    for (const start of openByPage.values()) intervals.push([Math.max(fromTs, start), toTs]);

    const valid = intervals.filter(([a,b]) => b > a).sort((x,y) => x[0] - y[0]);
    const merged = [];
    for (const interval of valid) {
      const last = merged[merged.length - 1];
      if (!last || interval[0] > last[1]) merged.push([...interval]);
      else last[1] = Math.max(last[1], interval[1]);
    }

    return {
      idleMs: merged.reduce((sum,[a,b]) => sum + (b-a), 0),
      periods: merged.length,
      rawIntervals: valid.length
    };
  }

  function fcrBurstStats(events) {
    const responses = new Map(
      events.filter(e => e.type === 'RESPONSE').map(e => [e.data?.reqId, e])
    );
    const byPage = new Map();

    for (const ev of events) {
      if (ev.type !== 'REQUEST') continue;
      if (!/fcresearch|qifcr/i.test(String(ev.data?.host || ''))) continue;
      if (!byPage.has(ev.pageId)) byPage.set(ev.pageId, []);
      byPage.get(ev.pageId).push(ev);
    }

    const bursts = [];
    for (const arr of byPage.values()) {
      arr.sort((a, b) => (a.ts || 0) - (b.ts || 0));
      let current = [];
      const push = () => {
        if (!current.length) return;
        const first = current[0];
        const last = current[current.length - 1];
        const durations = current
          .map(req => Number(responses.get(req.data?.reqId)?.data?.ms || 0))
          .filter(Boolean);
        bursts.push({
          at: first.at,
          count: current.length,
          launchSpanMs: Math.max(0, (last.ts || 0) - (first.ts || 0)),
          responseMedianMs: median(durations),
          responseMaxMs: durations.length ? Math.max(...durations) : 0,
          endpoints: new Set(current.map(x => endpointKey(x.data || {}))).size
        });
        current = [];
      };

      for (const req of arr) {
        if (!current.length || (req.ts || 0) - (current[current.length - 1].ts || 0) <= FCR_BURST_GAP_MS) {
          current.push(req);
        } else {
          push();
          current.push(req);
        }
      }
      push();
    }

    return bursts.sort((a, b) => b.count - a.count || b.responseMaxMs - a.responseMaxMs);
  }

  function fcrEndpointPresence(events) {
    const counts = new Map();
    for (const ev of events) {
      if (ev.type !== 'REQUEST') continue;
      if (!/fcresearch|qifcr/i.test(String(ev.data?.host || ''))) continue;
      const ep = endpointKey(ev.data || {}).replace(/^.*?\/BWU2\/results\//, '');
      counts.set(ep, (counts.get(ep) || 0) + 1);
    }
    return [...counts.entries()]
      .map(([endpoint, count]) => ({ endpoint, count }))
      .sort((a,b) => b.count - a.count || a.endpoint.localeCompare(b.endpoint));
  }

  function scriptEventStats(events) {
    const counts = new Map();
    for (const ev of events) {
      if (ev.type !== 'SCRIPT_EVENT') continue;
      const name = String(ev.data?.event || 'UNKNOWN');
      counts.set(name, (counts.get(name) || 0) + 1);
    }
    return [...counts.entries()]
      .map(([event, count]) => ({ event, count }))
      .sort((a, b) => b.count - a.count || a.event.localeCompare(b.event));
  }

  function handoffStats(events) {
    const byPage = new Map();
    for (const ev of events) {
      if (!byPage.has(ev.pageId)) byPage.set(ev.pageId, []);
      byPage.get(ev.pageId).push(ev);
    }
    const gaps = [];
    for (const arr of byPage.values()) {
      for (let i = 0; i < arr.length; i++) {
        const ev = arr[i];
        if (ev.type !== 'RESPONSE') continue;
        for (let j = i + 1; j < arr.length; j++) {
          const next = arr[j];
          const gap = (next.ts || 0) - (ev.ts || 0);
          if (gap > 5000) break;
          if (next.type === 'SCREEN') {
            gaps.push({
              gapMs: gap,
              endpoint: endpointKey(ev.data || {}),
              screen: next.data?.label || '',
              status: ev.data?.status || 0,
              at: ev.at
            });
            break;
          }
        }
      }
    }
    return gaps;
  }

  function buildSummary(events, fromTs, toTs, label = 'CURRENT CHUNK') {
    const responses = events.filter(e => e.type === 'RESPONSE');
    const requests = events.filter(e => e.type === 'REQUEST');
    const cancellations = events.filter(e => e.type === 'REQUEST_CANCELLED');
    const errors = events.filter(e =>
      ['REQUEST_ERROR', 'JS_ERROR', 'PROMISE_REJECTION', 'TRACER_ERROR'].includes(e.type) ||
      (e.type === 'RESPONSE' && (!e.data?.ok || Number(e.data?.status || 0) >= 400))
    );

    const workflowRequests = requests.filter(e => (e.data?.class || requestClass(e.data || {})) === 'workflow');
    const pollRequests = requests.filter(e => (e.data?.class || requestClass(e.data || {})) === 'poll');
    const backgroundRequests = requests.filter(e => (e.data?.class || requestClass(e.data || {})) === 'background');
    const duplicates = workflowRequests.filter(e => e.data?.duplicate);

    const userEnters = events.filter(e => e.type === 'ENTER' && e.data?.trusted).length;
    const scriptEnters = events.filter(e => e.type === 'ENTER' && !e.data?.trusted).length;
    const userClicks = events.filter(e => e.type === 'CLICK' && e.data?.trusted).length;
    const scriptClicks = events.filter(e => e.type === 'CLICK' && !e.data?.trusted).length;
    const screens = events.filter(e => e.type === 'SCREEN');

    const endpoints = endpointStats(events);
    const handoffs = handoffStats(events);
    const actionGaps = actionToRequestStats(events);
    const idle = idleStats(events, fromTs, toTs);
    const fcrBursts = fcrBurstStats(events);
    const fcrPresence = fcrEndpointPresence(events);
    const scriptEvents = scriptEventStats(events);

    const healthSamples = events.filter(e => e.type === 'HEALTH_SAMPLE');
    const eventLoopStalls = events.filter(e => e.type === 'EVENT_LOOP_STALL');
    const timerSuspends = events.filter(e => e.type === 'TIMER_SUSPEND');
    const longTasks = events.filter(e => e.type === 'LONG_TASK');
    const durationMs = Math.max(0, toTs - fromTs);
    const activeMs = Math.max(0, durationMs - idle.idleMs);
    const responseDurations = responses
      .filter(e => (e.data?.class || requestClass(e.data || {})) === 'workflow')
      .map(e => Number(e.data?.ms || 0));

    const tracerMeasuredMs = healthSamples.reduce((sum, e) =>
      sum +
      Number(e.data?.tracerAddMs || 0) +
      Number(e.data?.tracerFlushMs || 0) +
      Number(e.data?.tracerScreenMs || 0), 0
    );
    const mutationRecords = healthSamples.reduce((sum, e) => sum + Number(e.data?.mutationRecords || 0), 0);
    const mutationCallbacks = healthSamples.reduce((sum, e) => sum + Number(e.data?.mutationCallbacks || 0), 0);
    const maxEventLoopLag = Math.max(
      0,
      ...healthSamples.map(e => Number(e.data?.maxEventLoopLagMs || 0)),
      ...eventLoopStalls.map(e => Number(e.data?.ms || 0))
    );
    const maxLongTask = Math.max(0, ...longTasks.map(e => Number(e.data?.ms || 0)));

    const slow = [...responses]
      .filter(e => (e.data?.class || requestClass(e.data || {})) === 'workflow')
      .sort((a, b) => Number(b.data?.ms || 0) - Number(a.data?.ms || 0))
      .slice(0, 20);

    const slowHandoffs = handoffs
      .filter(x => x.gapMs >= 300)
      .sort((a, b) => b.gapMs - a.gapMs)
      .slice(0, 15);

    const slowActionGaps = actionGaps
      .filter(x => x.gapMs >= 25)
      .sort((a, b) => b.gapMs - a.gapMs)
      .slice(0, 15);

    const activeMinutes = activeMs / 60000;
    const mutationRate = activeMinutes > 0 ? Math.round(mutationRecords / activeMinutes) : 0;
    const tracerPct = activeMs > 0 ? (tracerMeasuredMs / activeMs * 100) : 0;

    const lines = [
      `BWU2 SUPER TRACE v${VERSION}`,
      `${label}`,
      `Session: ${activeSessionId}`,
      `From: ${new Date(fromTs).toISOString()}`,
      `To:   ${new Date(toTs).toISOString()}`,
      `Duration: ${(durationMs / 60000).toFixed(1)} min | Active: ${(activeMs / 60000).toFixed(1)} min | Idle union: ${(idle.idleMs / 60000).toFixed(1)} min (${idle.periods} merged periods; ${idle.rawIntervals} tab intervals)`,
      '',
      'ANALYSIS REQUEST',
      'Judge QoL-script efficiency and safety. Find slow backend calls, script-side delay after user input, duplicate work, polling/background noise, DOM churn, event-loop stalls, long tasks, excessive request fan-out, retry loops, and opportunities to shorten FCResearch/Sideline/MoveContainer workflows without risking inventory correctness.',
      '',
      'SUMMARY',
      `Events: ${events.length}`,
      `Requests: ${requests.length} | workflow ${workflowRequests.length} | poll ${pollRequests.length} | background ${backgroundRequests.length}`,
      `Responses: ${responses.length}`,
      `Errors: ${errors.length} | cancelled/aborted: ${cancellations.length}`,
      `Real duplicate workflow requests <=${DUP_WINDOW_MS}ms: ${duplicates.length}`,
      `Workflow network median: ${median(responseDurations)}ms | p95: ${percentile(responseDurations, 0.95)}ms`,
      `Trusted Enter→workflow request: median ${median(actionGaps.map(x => x.gapMs))}ms | p95 ${percentile(actionGaps.map(x => x.gapMs), 0.95)}ms | matched ${actionGaps.length}`,
      `Screen transitions: ${screens.length}`,
      `Enter actions: user ${userEnters} | script ${scriptEnters}`,
      `Clicks: user ${userClicks} | script ${scriptClicks}`,
      `Event-loop stalls >=${EVENT_LOOP_WARN_MS}ms: ${eventLoopStalls.length} | max ${round(maxEventLoopLag)}ms`,
      `Background/tab timer suspends excluded: ${timerSuspends.length}`,
      `Long tasks: ${longTasks.length} | max ${round(maxLongTask)}ms`,
      `DOM mutation activity observed: ${mutationRecords} records / ${mutationCallbacks} callbacks | ~${mutationRate}/active min`,
      `Tracer measured work: ${Math.round(tracerMeasuredMs)}ms | ~${tracerPct.toFixed(3)}% of active time`,
      `Script hook events: ${scriptEvents.reduce((s, x) => s + x.count, 0)}`,
      '',
      'ENDPOINTS',
      'class | count | median | p95 | max | err | dup | endpoint',
      ...endpoints.slice(0, 70).map(x =>
        `${x.class} | ${x.count} | ${x.medianMs}ms | ${x.p95Ms}ms | ${x.maxMs}ms | ${x.errors} | ${x.duplicates} | ${x.endpoint}`
      ),
      '',
      'FCRESEARCH REQUEST BURSTS',
      'requests | unique endpoints | launch span | response median | response max | started',
      ...fcrBursts.slice(0, 20).map(x =>
        `${x.count} | ${x.endpoints} | ${x.launchSpanMs}ms | ${x.responseMedianMs}ms | ${x.responseMaxMs}ms | ${x.at}`
      ),
      '',
      'FCRESEARCH ENDPOINT PRESENCE',
      'count | endpoint',
      ...fcrPresence.slice(0, 40).map(x => `${x.count} | ${x.endpoint}`),
      '',
      'SCRIPT EVENTS',
      ...scriptEvents.slice(0, 40).map(x => `${x.count} | ${x.event}`),
      '',
      'SLOWEST WORKFLOW REQUESTS',
      ...slow.map(e =>
        `${e.data?.ms || 0}ms | ${e.data?.status || 0} | ${e.data?.method || ''} ${endpointKey(e.data || {})} | ${e.at}`
      ),
      '',
      'SLOW USER ENTER→REQUEST HANDOFFS',
      ...slowActionGaps.map(x => `${x.gapMs}ms | ${x.action} -> ${x.endpoint} | ${x.crossedPage ? 'cross-page' : 'same-page'} | ${x.at}`),
      '',
      'SUSPECT RESPONSE→SCREEN HANDOFFS',
      '(Still correlation only; idle time is now measured separately.)',
      ...slowHandoffs.map(x => `${x.gapMs}ms | ${x.endpoint} -> ${x.screen} | HTTP ${x.status} | ${x.at}`),
      '',
      'REAL DUPLICATE WORKFLOW REQUESTS',
      ...duplicates.slice(-60).map(e =>
        `${e.at} | +${e.data?.previousGapMs ?? '?'}ms | ${e.data?.method || ''} ${endpointKey(e.data || {})}`
      ),
      '',
      'BACKGROUND / TAB TIMER SUSPENDS',
      ...timerSuspends.slice(-40).map(e => `${e.at} | ${clip(e.data, 500)}`),
      '',
      'CANCELLED / ABORTED REQUESTS',
      ...cancellations.slice(-60).map(e => `${e.at} | ${clip(e.data, 900)}`),
      '',
      'ERRORS',
      ...errors.slice(-80).map(e => `${e.at} | ${e.type} | ${clip(e.data, 900)}`)
    ];
    return lines.join('\n');
  }

  function buildFullDump(fromTs, toTs, label = 'CURRENT CHUNK') {
    const all = allEventsForSession(activeSessionId);
    const events = all.filter(e => (e.ts || 0) >= fromTs && (e.ts || 0) <= toTs);
    const summary = buildSummary(events, fromTs, toTs, label);
    return `${summary}\n\nRAW EVENTS (JSONL)\n${events.map(e => JSON.stringify(e)).join('\n')}\n`;
  }

  function filenameFor(fromTs, toTs, suffix = '') {
    const stamp = new Date(toTs).toISOString().replace(/[:.]/g, '-');
    return `BWU2_Super_Trace_${stamp}${suffix ? `_${suffix}` : ''}.txt`;
  }

  function downloadText(text, filename) {
    try {
      const blob = new Blob([text], { type: 'text/plain;charset=utf-8' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = filename;
      a.style.display = 'none';
      (document.body || document.documentElement).appendChild(a);
      a.click();
      a.remove();
      setTimeout(() => URL.revokeObjectURL(url), 5000);
      return true;
    } catch (error) {
      add('TRACER_ERROR', { area: 'download', error: clip(error?.message || error, 800) });
      return false;
    }
  }

  function acquireDumpLock() {
    const id = `${pageId}:${epoch()}`;
    const current = parseJSON(gmGet(LOCK_KEY, ''), null);
    if (current?.at && epoch() - current.at < 10000) return null;
    gmSet(LOCK_KEY, JSON.stringify({ id, at: epoch() }));
    const verify = parseJSON(gmGet(LOCK_KEY, ''), null);
    return verify?.id === id ? id : null;
  }

  function releaseDumpLock(id) {
    const current = parseJSON(gmGet(LOCK_KEY, ''), null);
    if (current?.id === id) gmDelete(LOCK_KEY);
  }

  function dumpCurrentChunk({ advance = false } = {}) {
    // MANUAL ONLY. Never call this from the 30-minute timer.
    syncSessionIfChanged();
    const lock = acquireDumpLock();
    if (!lock) return false;
    try {
      meta = loadMeta();
      const fromTs = Number(meta.chunkStartedAt || meta.sessionStartedAt || epoch());
      const toTs = epoch();
      const text = buildFullDump(fromTs, toTs, 'MANUAL DUMP');
      const ok = downloadText(text, filenameFor(fromTs, toTs));
      add('MANUAL_DUMP', { fromTs, toTs, ok, eventsApprox: text.split('\n').length });
      if (advance) {
        meta.lastDumpFrom = fromTs;
        meta.lastDumpTo = toTs;
        meta.chunkStartedAt = toTs + 1;
        meta.nextDumpAt = toTs + AUTO_DUMP_MS;
        saveMeta(meta);
      }
      return ok;
    } finally {
      releaseDumpLock(lock);
    }
  }

  function snapshotCurrentChunk({ advance = true } = {}) {
    // SAFE AUTO PATH: build + store internally only.
    // Absolutely no anchor click / browser download / focus change here.
    syncSessionIfChanged();
    const lock = acquireDumpLock();
    if (!lock) return false;

    try {
      meta = loadMeta();
      const fromTs = Number(meta.chunkStartedAt || meta.sessionStartedAt || epoch());
      const toTs = epoch();
      const filename = filenameFor(fromTs, toTs);
      const text = buildFullDump(fromTs, toTs, 'AUTO 30-MINUTE SNAPSHOT');

      const count = archiveSnapshot({
        sessionId: activeSessionId,
        fromTs,
        toTs,
        filename,
        savedAt: toTs,
        text
      });

      if (advance) {
        meta.lastDumpFrom = fromTs;
        meta.lastDumpTo = toTs;
        meta.chunkStartedAt = toTs + 1;
        meta.nextDumpAt = toTs + AUTO_DUMP_MS;
        saveMeta(meta);
      }

      add('AUTO_SNAPSHOT', {
        fromTs,
        toTs,
        stored: true,
        pendingSnapshots: count,
        eventsApprox: text.split('\n').length
      });
      return true;
    } catch (error) {
      add('TRACER_ERROR', { area: 'snapshot', error: clip(error?.message || error, 800) });
      return false;
    } finally {
      releaseDumpLock(lock);
    }
  }

  function downloadOldestSnapshot() {
    const list = loadSnapshots();
    if (!list.length) return false;

    const snapshot = list[0];
    const ok = downloadText(snapshot.text, snapshot.filename);
    if (!ok) return false;

    // Keep one recovery copy in case the browser download is cancelled/misplaced.
    gmSet(LAST_EXPORT_KEY, JSON.stringify(snapshot));
    list.shift();
    saveSnapshots(list);
    toast(`Trace downloaded • ${list.length} pending`);
    updateUi();
    return true;
  }

  function downloadPreviousExport() {
    const snapshot = parseJSON(gmGet(LAST_EXPORT_KEY, ''), null);
    if (!snapshot?.text || !snapshot?.filename) return false;
    return downloadText(snapshot.text, snapshot.filename.replace(/\.txt$/i, '_REDOWNLOAD.txt'));
  }

  async function copyChatSummary() {
    syncSessionIfChanged();
    meta = loadMeta();
    const fromTs = Number(meta.chunkStartedAt || meta.sessionStartedAt || epoch());
    const toTs = epoch();
    const all = allEventsForSession(activeSessionId).filter(e => (e.ts || 0) >= fromTs && (e.ts || 0) <= toTs);
    const text = buildSummary(all, fromTs, toTs, 'CURRENT CHAT SUMMARY');
    try {
      await navigator.clipboard.writeText(text);
      toast('Summary copied');
      return true;
    } catch {}
    try {
      const ta = document.createElement('textarea');
      ta.value = text;
      Object.assign(ta.style, { position: 'fixed', left: '-99999px', top: '0' });
      document.documentElement.appendChild(ta);
      ta.select();
      const ok = document.execCommand('copy');
      ta.remove();
      if (ok) toast('Summary copied');
      return ok;
    } catch {
      toast('Copy failed');
      return false;
    }
  }

  function newSession() {
    flushPage();
    const oldSession = activeSessionId;
    const next = defaultMeta();
    saveMeta(next);
    meta = next;
    activeSessionId = next.sessionId;
    pageEvents = [];
    pageStoreKey = `${PAGE_PREFIX}${activeSessionId}:${pageId}`;
    recentRequestSignatures = new Map();
    lastForegroundActivityAt = epoch();
    idleSince = 0;
    resetHealthWindow();
    try { gmSet(GLOBAL_ACTION_KEY, ''); } catch {}
    add('SESSION_START', { previousSession: oldSession, href: location.href, version: VERSION });
    sampleScreen(true);
    toast('New trace session');
  }

  function cleanupOldSessions() {
    try {
      const current = loadMeta().sessionId;
      const cutoff = epoch() - 24 * 60 * 60 * 1000;
      for (const key of gmKeys()) {
        if (!key.startsWith(PAGE_PREFIX) || key.includes(`${PAGE_PREFIX}${current}:`)) continue;
        const page = parseJSON(gmGet(key, ''), null);
        if (!page?.updatedAt || page.updatedAt < cutoff) gmDelete(key);
      }
    } catch {}
  }

  function autoDumpTick() {
    if (!settings.autoDump) return;
    syncSessionIfChanged(true);
    meta = loadMeta();
    if (epoch() < Number(meta.nextDumpAt || 0)) return;
    snapshotCurrentChunk({ advance: true });
  }

  function currentCounts() {
    syncSessionIfChanged();
    meta = loadMeta();
    const fromTs = Number(meta.chunkStartedAt || meta.sessionStartedAt || 0);

    const all = allEventsForSession(activeSessionId).filter(e => (e.ts || 0) >= fromTs);

    return {
      events: all.length,
      requests: all.filter(e => e.type === 'REQUEST').length,
      errors: all.filter(e => ['REQUEST_ERROR', 'JS_ERROR', 'PROMISE_REJECTION', 'TRACER_ERROR'].includes(e.type) || (e.type === 'RESPONSE' && (!e.data?.ok || Number(e.data?.status || 0) >= 400))).length,
      cancelled: all.filter(e => e.type === 'REQUEST_CANCELLED').length,
      duplicates: all.filter(e => e.type === 'REQUEST' && e.data?.duplicate && (e.data?.class || requestClass(e.data || {})) === 'workflow').length,
      stalls: all.filter(e => e.type === 'EVENT_LOOP_STALL').length,
      suspends: all.filter(e => e.type === 'TIMER_SUSPEND').length
    };
  }

    if (!document.documentElement) return;
    const el = document.createElement('div');
    el.textContent = text;

      position: 'fixed', right: '12px', top: '48px', zIndex: 2147483647,

      boxShadow: '0 2px 8px rgba(0,0,0,.25)'
    });
    document.documentElement.appendChild(el);
    setTimeout(() => el.remove(), 1600);
  }

      }

    auto.addEventListener('change', () => {
      settings.autoDump = auto.checked;
      saveSettings();

      settings.includeRecognizedValues = values.checked;

    });

  function updateUi() {
    if (!uiRoot) return;
    const stat = uiRoot.querySelector('.bwu2st-stats');
    const pill = uiRoot.querySelector('.bwu2st-pill');
    if (!stat || !pill) return;
    meta = loadMeta();
    const c = currentCounts();
    const elapsedMin = Math.max(0, (epoch() - Number(meta.chunkStartedAt || epoch())) / 60000);
    const untilMin = Math.max(0, (Number(meta.nextDumpAt || epoch()) - epoch()) / 60000);
    const pending = pendingSnapshotCount();
    const pendingBtn = uiRoot.querySelector('[data-a="pending"]');
    if (pendingBtn) pendingBtn.textContent = `Pending snapshots (${pending})`;
    stat.textContent = `GLOBAL SESSION\nChunk ${elapsedMin.toFixed(1)}m\nEvents ${c.events} | Req ${c.requests}\nErrors ${c.errors} | Cancel ${c.cancelled} | Dup ${c.duplicates}\nLoop stalls ${c.stalls} | Tab suspends ${c.suspends}\nSilent snapshot ${settings.autoDump ? `in ${untilMin.toFixed(1)}m` : 'OFF'}\nPending exports ${pending}`;
    pill.textContent = c.errors ? `TRACE GLOBAL !${c.errors}` : `TRACE ● GLOBAL ${Math.floor(elapsedMin)}m`;
  }

  function bootUi() {
    if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', mountUi, { once: true });
    else mountUi();
  }

  function installScriptEventBus() {
    // @grant none page scripts can call: window.BWU2Trace('EVENT_NAME', { ...data })
    // Sandboxed/GM scripts can call: window.postMessage({ __BWU2_TRACE__: true, type: 'EVENT_NAME', data: {...} }, '*')
    try {
      W.BWU2Trace = function(type, data = {}) {
        markForegroundActivity(`script:${clip(type, 60)}`);
        add('SCRIPT_EVENT', { event: clip(type, 120), data: redactObject(data) });
      };
    } catch {}

    W.addEventListener('message', event => {
      try {
        const msg = event?.data;
        if (!msg || msg.__BWU2_TRACE__ !== true) return;
        markForegroundActivity(`script:${clip(msg.type || 'MESSAGE', 60)}`);
        add('SCRIPT_EVENT', { event: clip(msg.type || 'MESSAGE', 120), data: redactObject(msg.data || {}) });
      } catch {}
    }, true);
  }

    if (!downloadOldestSnapshot()) toast('No pending snapshots');
  }); } catch {}
  try { GM_registerMenuCommand('BWU2 Trace: Re-download previous export', () => {

    host: location.host,
    path: location.pathname,
    title: clean(document.title),

  });
